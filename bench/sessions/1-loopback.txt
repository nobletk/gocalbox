# Session 1: loopback, Linux, on PC A (X1, X2, X3, X5, X6 of the benchmark plan).
# Run on PC A without SERVER_SSH; the client and the server share the machine.
# X1's baselines (tusd, SFTP and scp) need tusd and OpenSSH (bench/README.md); run them with
# the rest of X1, CELLS=x1-, so that the engine and the baselines share the session.
#
# PRE_M6 names the commit before M6 (bench/README.md says how to find it).

server default
server pre-m6 SERVER_REV=$PRE_M6
server h2-bench BUILD_TAGS=bench
server h2-128k BUILD_TAGS=bench GOCALBOX_BENCH_H2_STREAM_WINDOW=128K
server h2-448k BUILD_TAGS=bench GOCALBOX_BENCH_H2_STREAM_WINDOW=448K
server h2-1m BUILD_TAGS=bench GOCALBOX_BENCH_H2_STREAM_WINDOW=1M GOCALBOX_BENCH_H2_CONN_WINDOW=16M
server tusd KIND=tusd
server sshd KIND=sshd

# X1's baselines. The tus core protocol sends one request at a time, so the engine is

# default cipher; the batch is 100 files in one sftp session.

cell x1-engine-p1-h2-64M  default engine   transport=https-h2 size=64M parallel=1

# X2: what the guarantees cost. Auditing, in process: the default, every part recorded,
# and changes answered before their records are on disk. Then M6's staging and flush:
# the single-request upload before M6 against now.
cell x2-audit-default-1G - inprocess size=1G
cell x2-audit-parts-1G   - inprocess size=1G audit-parts=1
cell x2-audit-nowait-1G  - inprocess size=1G audit-wait=0
cell x2-audit-default-1M - inprocess size=1M files=200
cell x2-audit-parts-1M   - inprocess size=1M files=200 audit-parts=1
cell x2-audit-nowait-1M  - inprocess size=1M files=200 audit-wait=0
